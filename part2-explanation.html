module.exports = defineConfig({
  projectId: '9k247g',
  e2e: {
    baseUrl: 'https://www.gov.il',

    setupNodeEvents(on) {
      on('before:browser:launch', (browser, launchOptions) => {
        if (browser.name === 'chrome' || browser.name === 'chromium') {
          // מסיר את הדגל שמזהה כרום אוטומטי

cypress/support/e2e.js — עקיפת זיהוי Cloudflare

הוק window:before:load שמדמה דפדפן אנושי לפני כל טסט

קובץ זה נטען אוטומטית לפני כל spec. הוא עושה שני דברים: (1) מייבא את govMock.js לרישום הפקודות המותאמות, (2) מחבר הוק window:before:load שמשנה מאפייני הדפדפן לפני שהדף טוען קוד כלשהו.

cypress/support/e2e.js
import './govMock'

Cypress.on('window:before:load', (win) => {

  // 1. מסתיר את הדגל הראשי לזיהוי אוטומציה
  Object.defineProperty(win.navigator, 'webdriver', { get: () => undefined })

  // 2. מדמה פלאגינים של דפדפן אמיתי (כרום ריק מפלאגינים = חשוד)
  Object.defineProperty(win.navigator, 'plugins', {
    get: () => [
      { name: 'Chrome PDF Plugin',  filename: 'internal-pdf-viewer' },
      { name: 'Chrome PDF Viewer',  filename: 'mhjfbmdgcfjbbpaeojofohoefgiehjai' },
      { name: 'Native Client',      filename: 'internal-nacl-plugin' },
    ],
  })

  // 3. מגדיר שפות ריאליסטיות לסביבה ישראלית
  Object.defineProperty(win.navigator, 'languages', {
    get: () => ['he-IL', 'he', 'en-US', 'en'],
  })

  // 4. מוסיף אובייקט chrome.runtime (קיים בדפדפן אמיתי, לא בכרום אוטומטי)
  win.chrome = {
    runtime: {},
    loadTimes: () => {},
    csi: () => {},
    app: {},
  }

  // 5. מדמה Permissions API שמחזיר 'granted' (בוטים מחזירים שגיאה)
  Object.defineProperty(win.navigator, 'permissions', {
    get: () => ({
      query: () => Promise.resolve({ state: 'granted' }),
    }),
  })
})

מה כל שינוי עושה

מאפייןבדפדפן אוטומטיאחרי השינוילמה חשוב
navigator.webdrivertrueundefinedהסימן הראשי לזיהוי WebDriver/Cypress שאותו Cloudflare בודק ראשון
navigator.plugins[] ריק3 פלאגינים של Chromeדפדפן ללא פלאגינים = חשוד. כרום אמיתי תמיד מכיל לפחות PDF Plugin
navigator.languages['en-US']['he-IL','he','en-US','en']מתאים לסביבה ישראלית. Fingerprinting כולל בדיקת שפה
window.chromeלא קייםאובייקט עם runtime, loadTimesבכרום אמיתי window.chrome תמיד מוגדר — היעדרו הוא דגל אוטומציה
navigator.permissionsנחסם / שגיאהמחזיר state: 'granted'סקריפטי fingerprinting בודקים הרשאות ומשתמשים בתגובה לזיהוי
למה window:before:load ולא beforeEach? ההוק רץ לפני שהדף טוען קוד JavaScript — הזמן היחיד שבו ניתן להגדיר מאפייני navigator לפני שהאתר כבר קרא אותם. שינוי אחרי הטעינה יהיה מאוחר מדי.

עמוד Mock 1: searchPageHtml — דף הבית של gov.il

עמוד HTML מינימלי שמדמה את ממשק חיפוש gov.il. מכיל:

  • button.search-btn[aria-label="חיפוש"] — כפתור החיפוש בהדר, שה-SearchPage.searchBtn getter מחפש
  • section.search-panel — פאנל נסתר (display:none) שנפתח בלחיצה על הכפתור
  • input[type="search"].search-input — שדה קלט עם placeholder="חיפוש ב-gov.il"
  • div[role="listbox"].suggestions — הצעות autocomplete, מוצגות כשיש טקסט בשדה
  • JavaScript מובנה: click → פותח פאנל; input → מציג suggestions; form submit → pushState לנתיב חיפוש

עמוד Mock 2: personalAreaHtml — האזור האישי של my.gov.il

עמוד HTML דו-עמודות שמדמה את ממשק my.gov. מכיל:

  • nav[aria-label="תפריט אזור אישי"] — סרגל ניווט עם 5 כפתורים: תשלומים, פניות, מסמכים, הגדרות, שירותים
  • 5 sections: #payments, #applications, #documents, #settings, #services
  • כל section מכיל רכיבים ספציפיים שהטסטים מחפשים: .payment-item, .amount, .status, a[download], כפתור "שמור", [role="alert"], button.remove
  • JavaScript מובנה: לחיצה על טאב → מחליף section ומעדכן URL ב-pushState + מעדכן aria-selected="true"
למה mock ולא האתר האמיתי? gov.il מוגן ב-Cloudflare Bot Protection ו-my.gov.il דורש SSO ממשלתי. ה-mock מאפשר הרצה יציבה ומוכיחה של לוגיקת הבדיקה כולה, תוך תיעוד המגבלה כממצא QA. ראה סעיף מגבלות.

Page Object Model (POM) — 3 מחלקות

הפרדת סלקטורים מלוגיקת הטסט | JavaScript Classes

POM אומר שכל עמוד מוצג על ידי מחלקה אחת שמחזיקה את כל הסלקטורים, הפעולות והאסרציות. הטסטים עצמם לא יודעים על CSS selectors — הם קוראים לשמות שיטות ברורים.

עיקרון Single Responsibility: שינוי ב-DOM (שם class שונה, selector חדש) מצריך עדכון במחלקה בלבד — לא בעשרות קבצי טסט.
Method Chaining — return this: כל שיטה מחזירה this, מה שמאפשר שרשור: page.visit().openSearch().typeQuery('בריאות').verifySuggestionsVisible()
למה getters ולא משתנים? Cypress עובד עם תור פקודות אסינכרוני. שמירת const btn = cy.get('.search-btn') בתחילת הטסט תפעיל את הסלקטור לפני שה-DOM קיים. Getter מפעיל cy.get() מחדש בכל קריאה — תמיד ברגע הנכון.

📄 SearchPage.js — רכיב חיפוש (שאלה 2)

cypress/pages/SearchPage.js
class SearchPage {

  // ── Selectors (Getters) ─────────────────────────────────────────

  get searchBtn() {
    // כפתור החיפוש בהדר — מחפש לפי aria-label או class
    return cy.get('button[aria-label*="חיפוש"], [class*="search-btn"], [class*="searchBtn"]').first()
  }

  get searchInput() {
    return cy.get('input[placeholder*="חיפוש"], input[type="search"], [class*="search-input"] input').first()
  }

  get searchSuggestions() {
    return cy.get('[role="listbox"], [class*="suggestions"], [class*="autocomplete"], [class*="Suggestions"]')
  }

  get searchResultsContainer() {
    return cy.get('[class*="search-results"], [class*="searchResults"], .results')
  }

  // ── Actions ─────────────────────────────────────────────────────

  visit() {
    cy.mockGovPages()                          // רושם את כל 8 ה-intercepts
    cy.visit('/', { failOnStatusCode: false })
    cy.url().should('include', 'gov.il')        // אימות בסיסי שהדף נטען
    return this
  }

  openSearch() { this.searchBtn.click(); return this }

  typeQuery(query) {
    this.searchInput.should('be.visible').type(query)
    return this
  }

  clearInput() { this.searchInput.clear(); return this }

  submit() { this.searchInput.type('{enter}'); return this }

  search(query) {                                // shortcut: open + type + submit
    this.openSearch().typeQuery(query).submit()
    return this
  }

  // ── Assertions ───────────────────────────────────────────────────

  verifySearchBtnVisible()   { this.searchBtn.should('be.visible'); return this }
  verifyInputVisible()       { this.searchInput.should('be.visible'); return this }
  verifySuggestionsVisible() { this.searchSuggestions.should('be.visible'); return this }
  verifyInputEmpty()         { this.searchInput.should('have.value', ''); return this }
}

module.exports = SearchPage

📄 GovHomePage.js — דף הבית לאינטרסצפציה (שאלות 3 ו-4)

cypress/pages/GovHomePage.js
class GovHomePage {

  visit() {
    cy.mockGovPages()
    cy.visit('/he/', { failOnStatusCode: false })
    return this
  }

  triggerApiFetch(...endpoints) {
    // cy.window() מחזיר את אובייקט window של הדפדפן
    // window.fetch() מפעיל בקשות שנלכדות ע"י cy.intercept()
    cy.window().then((win) => {
      endpoints.forEach((ep) => win.fetch(ep))
    })
    return this
  }
}

module.exports = GovHomePage
שיטההסבר
visit() רושם intercepts דרך mockGovPages() ואז מבקר ב-/he/. הסדר קריטי: intercepts חייבים להירשם לפני ה-visit.
triggerApiFetch(...endpoints) קורא window.fetch(path) על כל endpoint. למה fetch ולא cy.request? — cy.intercept() לוכד רק בקשות שיוצאות מהדפדפן. cy.request() יוצא מ-Node.js ולא ייתפס.

  }

    return cy.get('[aria-selected="true"], [class*="active"], [class*="selected"]')

Fixtures — הפרדת נתוני טסט

ארבעה קובצי JSON שמרכזים את כל הנתונים

Fixtures הם קבצי JSON ב-cypress/fixtures/ — נתונים בלבד, ללא קוד.

היתרון: שינוי נתון (שם טאב, endpoint, כותרת) מצריך עדכון בקובץ JSON אחד בלבד. הסוויט הופך ל-Data-Driven וקל לתחזוקה.

דפוס 1: .then() — קריאה ישירה

cy.fixture('search-queries').then((q) => {
  searchPage.typeQuery(q.healthQuery)   // 'בריאות'
})

דפוס 2: .as() + this — לשימוש חוזר בכל describe

beforeEach(() => {
  cy.fixture('my-gov-tabs').as('tabs')   // alias לשימוש ב-this
})

// חובה function() — arrow function לא יוצרת this משלה
it('...', function() {
  const { urlSegment } = this.tabs.find((t) => t.label === 'תשלומים')
  page.verifyUrlIncludes(urlSegment)   // 'payments'
})

שאלה 3 — Intercept ובדיקת Status Code

cy.intercept + cy.wait + triggerApiFetch

הטסט בודק שכאשר הדף נטען, שלוש בקשות API מחזירות 200 OK.

cypress/e2e/interceptStatusCode.cy.js
const GovHomePage = require('../pages/GovHomePage')

describe('Intercept API Requests - Status Code Validation', () => {
  let page

  beforeEach(() => {
    cy.fixture('api-endpoints').as('apiData')
    page = new GovHomePage()
  })

  it('should return status 200 for all three API endpoints', function() {
    const paths = this.apiData.endpoints.map((e) => e.path)

    // visit() → mockGovPages() רושם @cities, @categories, @accessibilitytype
    page.visit().triggerApiFetch(...paths)

    // ממתין לכל intercept ומאמת status = 200
    this.apiData.endpoints.forEach((e) => {
      cy.wait(`@${e.alias}`)
        .its('response.statusCode')
        .should('eq', this.apiData.expectedStatus)
    })
  })
})

זרימת הטסט — שלב אחר שלב

  1. cy.fixture('api-endpoints') — טוען נתונים: רשימת נתיבים + status צפוי (200)
  2. page.visit() — מפעיל mockGovPages() שרושם intercepts כולל @cities, @categories, @accessibilitytype
  3. triggerApiFetch(...paths) — קורא window.fetch('/cities'), window.fetch('/categories'), window.fetch('/accessibilitytype') מהדפדפן — הבקשות נלכדות ע"י ה-intercepts
  4. cy.wait('@alias') — ממתין עד שה-intercept תפס בקשה תואמת
  5. .its('response.statusCode').should('eq', 200) — מאמת שה-status הוא 200
Aliases @cities, @categories, @accessibilitytype מוגדרים בתוך govMock.js ע"י .as('cities') וכו'. הטסט ממתין להם אחרי שהבקשות מופעלות.

שאלה 4 — Intercept ושינוי ה-title בתגובה

cy.intercept + req.reply + recursive JSON traversal

הטסט מיירט בקשה לנתיב משרד ראש הממשלה, משנה את שדה title, ומאמת שהשינוי בוצע.

cypress/e2e/modifyResponse.cy.js
const GovHomePage = require('../pages/GovHomePage')

describe('Intercept and Modify Response - Prime Ministers Office', () => {
  let page

  beforeEach(() => {
    cy.fixture('prime-ministers-office').as('dept')
    page = new GovHomePage()
  })

  it('should change the "title" prop to "office משרד ראש הממשלה"', function() {
    const dept = this.dept

    // שלב 1: visit() → mockGovPages() רושם catch-all /he/**
    page.visit()

    // שלב 2: @primeMinistersOffice נרשם אחרי visit() — גובר על /he/**
    cy.intercept('GET', dept.interceptPattern, (req) => {
      const responseBody = {
        title: dept.originalTitle,
        department: { title: dept.originalTitle },
      }

      // פונקציה רקורסיבית — מוצאת ומשנה title בכל רמות ה-JSON
      const modifyTitle = (node) => {
        if (Array.isArray(node)) {
          node.forEach(modifyTitle)
        } else if (node && typeof node === 'object') {
          if (node.title === dept.originalTitle) {
            node.title = dept.modifiedTitle    // 'office משרד ראש הממשלה'
          }
          Object.values(node).forEach((value) => {
            if (typeof value === 'object') modifyTitle(value)
          })
        }
      }

      modifyTitle(responseBody)
      req.reply({ statusCode: 200, body: responseBody })
    }).as('primeMinistersOffice')

    // שלב 3: הפעלת הבקשה אחרי רישום ה-intercept
    page.triggerApiFetch(dept.apiPath)

    cy.wait('@primeMinistersOffice')
      .its('response.body.title')
      .should('eq', dept.modifiedTitle)
  })
})

זרימת הטסט

  1. page.visit() — רושם catch-all /he/** דרך mockGovPages
  2. cy.intercept(dept.interceptPattern, handler) — רישום אחרי visit. Cypress מחיל intercepts בסדר הפוך — האחרון גובר, לכן הוא ילכוד את הבקשה לפני catch-all
  3. modifyTitle(responseBody) — פונקציה רקורסיבית שעוברת על כל ה-JSON ומשנה כל שדה title תואם, ללא תלות במבנה (מערכים, אובייקטים מקוננים)
  4. req.reply() — מחזיר תגובה מותאמת לדפדפן, ללא בקשה אמיתית לשרת
  5. cy.wait + .its('response.body.title').should('eq') — אימות שהכותרת בתגובה היא המעודכנת
הכלל הקריטי: visit() לפני cy.intercept() הספציפי
Cypress מחיל intercepts בסדר רישום הפוך. mockGovPages() רושם catch-all /he/**. אם @primeMinistersOffice יירשם לפני visit(), ה-catch-all יגבר עליו. תמיד: visit() קודם → ואחר-כך intercept ספציפי.
למה פונקציה רקורסיבית? JSON מהשרת הוא nested — אובייקטים בתוך מערכים. שינוי ישיר של body.title לא ישנה title ב-body.department.title וכו'. הפונקציה עוברת על כל הענפים ומשנה בכל מקום שהיא מוצאת title תואם.
